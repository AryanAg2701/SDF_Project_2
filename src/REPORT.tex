\section*{Feature Added}
\begin{itemize}
    \item Introduced \texttt{try\{\} with\{\}} block for structured exception handling.
    \item Added two exceptions:
    \begin{itemize}
        \item \textbf{DivisionByZero}: Raised when division by zero occurs.
        \item \textbf{GenericException}: Raised when an undefined operation is performed (e.g., adding two values of different types).
    \end{itemize}
\end{itemize}

\section*{Changes Made}

\subsection*{1. Lexer}
\begin{itemize}
    \item Added new tokens:
    \begin{itemize}
        \item \texttt{try}
        \item \texttt{with}
        \item \texttt{\{}
        \item \texttt{\}}
        \item \texttt{DivisionByZero}
        \item \texttt{GenericException}
    \end{itemize}
\end{itemize}

\subsection*{2. Syntax}
\begin{itemize}
    \item Added a new AST node: \texttt{TryWith()}.
\end{itemize}

\subsection*{3. Parser}
\begin{itemize}
    \item Added grammar rules to handle the \texttt{try\{\} with\{\}} block.
\end{itemize}

\subsection*{4. Type Checking}
\begin{itemize}
    \item Added type checking for \texttt{TryWith()}.
    \item We only check the type of the expression in the with block if there is an error in the try block expression; otherwise, the machine handles it.
\end{itemize}

\subsection*{5. Machine}
\begin{itemize}
    \item Introduced a new machine value:
    \begin{itemize}
        \item \texttt{MException} of type \texttt{string} to hold the exception name.
    \end{itemize}
    \item Added a new instruction:
    \begin{itemize}
        \item \texttt{IHandle} to handle the \texttt{try\{\} with\{\}} block.
    \end{itemize}
    \item When an exception occurs:
    \begin{itemize}
        \item An \texttt{MException} with the exception name is pushed onto the machine stack when we encounter some behavior that is undefined.
    \end{itemize}
\end{itemize}

\section*{Correct Cases}

\begin{itemize}
  \item \texttt{try \{42\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{try \{3 * 5 + (if 5 < 6 then 10 else 100)\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{try \{21/6\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{try \{21/0\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{let multiplyby = fun f (n : int) : int is if n = 0 then 0 else 2*n;;}

        \texttt{try \{multiplyby 5\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{let fact = fun f (n : int) : int is if n = 0 then 1 else n * f (n-1);;}

        \texttt{try \{fact 5\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{let primedividebyn = fun f (n : int) : int is 333/n;;}

        \texttt{try \{primedividebyn 5\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{let primedividebyn = fun f (n : int) : int is 333/n;;}

        \texttt{try \{primedividebyn 0\} with \{ | DivisionByZero -> 100 \};;}
  \item \texttt{let safeprimedividebyn = fun f (n : int) : int is if n = 0 then 0 else 333/n;;}

        \texttt{try \{safeprimedividebyn 0\} with \{ | DivisionByZero -> 100 \};;}
  \item \texttt{let safeprimedividebyn = fun f (n : int) : int is if n = 0 then 0 else 333/n;;}

        \texttt{try \{safeprimedividebyn true\} with \{ | GenericException -> 100 \};;}
  \item \texttt{try \{2 + true\} with \{ | GenericException -> 100 \};;}
\end{itemize}

\section*{Malfunction Test Cases}

        \texttt{try \{safe\_primedividebyn true\} with \{ | DivisionByZero -> 100 \};;}

        \texttt{try \{multiplyby2 \textbackslash 5\} with \{ | DivisionByZero -> 0 \};;}
  \item \texttt{try \{2 + true\} with \{ | GenericException -> 100, DivisionByZero -> -100 \};;}
\end{itemize}
